// Client provides methods for interacting with the Aqua Temp API.
type Client interface {
        // Login authenticates with the Aqua Temp API.
        Login(ctx context.Context) error

        // Logout clears the authentication token.
        Logout(ctx context.Context) error

        // GetDevices returns a list of all devices.
        GetDevices(ctx context.Context) ([]Device, error)

        // GetDevice returns a specific device by code.
        GetDevice(ctx context.Context, deviceCode string) (*Device, error)

        // GetDeviceData returns the current sensor data for a device.
        GetDeviceData(ctx context.Context, deviceCode string) (*DeviceData, error)

        // SetTemperature sets the target temperature for a device.
        SetTemperature(ctx context.Context, deviceCode string, temp float64) error

        // SetMode sets the operation mode for a device.
        SetMode(ctx context.Context, deviceCode string, mode OperationMode) error

        // SetPower turns a device on or off.
        SetPower(ctx context.Context, deviceCode string, on bool) error

        // SetFanSpeed sets the fan speed for a device.
        SetFanSpeed(ctx context.Context, deviceCode string, speed FanSpeed) error

        // Control sends a compound control command to a device.
        Control(ctx context.Context, command ControlCommand) error

        // IsAuthenticated returns true if the client has a valid token.
        IsAuthenticated() bool

        // Config returns the client configuration.
        Config() *Config
}

// client is the default implementation of Client.
type client struct {
        config *Config
        auth   auth.AuthManager
        http   transport.HTTPClient
}

// NewClient creates a new Aqua Temp client with the given options.
func NewClient(opts ...Option) (Client, error) {
        // Start with default config
        config := DefaultConfig()

        // Apply environment config first
        envConfig := LoadConfigFromEnv()
        WithConfig(envConfig)(config)

        // Load config file (AQUATEMP_CONFIG env var or default path)
        cfgPath := config.ConfigPath
        if cfgPath == "" {
                cfgPath = DefaultConfigPath()
        }
        if cfgPath != "" {
                if fileCfg, err := LoadConfigFile(cfgPath); err == nil {
                        WithConfig(fileCfg)(config)
                }
        }

        // Apply provided options (highest priority)
        for _, opt := range opts {
                opt(config)
        }

        // Create HTTP transport
        httpClient := transport.NewClient(
                config.GetBaseURL(),
                config.Timeout,
                config.RetryCount,
                config.Debug,
        )

        // Create auth manager
        authManager := auth.NewManager(
                httpClient,
                string(config.APIVersion),
                config.Username,
                config.Password,
        )

        return &client{
                config: config,
                auth:   authManager,
                http:   httpClient,
        }, nil
}

// Login authenticates with the Aqua Temp API.
func (c *client) Login(ctx context.Context) error {
        if err := c.config.Validate(); err != nil {
                return err
        }

        _, err := c.auth.Authenticate(ctx)
        if err != nil {
                return &AuthError{
                        Code:      -1,
                        Message:   "login failed",
                        Retryable: false,
                        Err:       ErrInvalidCredentials,
                }
        }
        return nil
}

// Logout clears the authentication token.
func (c *client) Logout(ctx context.Context) error {
        c.auth.Clear()
        return nil
}

// GetDevices returns a list of all devices.
func (c *client) GetDevices(ctx context.Context) ([]Device, error) {
        if err := c.ensureAuth(ctx); err != nil {
                return nil, err
        }

        req := api.NewDeviceListRequest()
        body, err := c.http.Post(ctx, api.EndpointDeviceList, req, c.auth.FormatTokenHeader())
        if err != nil {
                return nil, fmt.Errorf("failed to get device list: %w", err)
        }

        var resp api.DeviceListResponse
        if err := json.Unmarshal(body, &resp); err != nil {
                return nil, fmt.Errorf("failed to parse device list response: %w", err)
        }

        if !resp.IsSuccess() {
                return nil, &APIError{
                        ErrorCode: resp.ErrorCode,
                        Message:   resp.ErrorMessage,
                }
        }

        devices := make([]Device, len(resp.ObjectResult))
        for i, d := range resp.ObjectResult {
                devices[i] = deviceInfoToDevice(&d)
        }

        return devices, nil
}

// GetDevice returns a specific device by code.
func (c *client) GetDevice(ctx context.Context, deviceCode string) (*Device, error) {
        devices, err := c.GetDevices(ctx)
        if err != nil {
                return nil, err
        }

        for _, d := range devices {
                if d.Code == deviceCode {
                        return &d, nil
                }
        }

        return nil, &DeviceError{
                DeviceCode: deviceCode,
                Operation:  "get",
                Err:        ErrDeviceNotFound,
        }
}

// GetDeviceData returns the current sensor data for a device.
func (c *client) GetDeviceData(ctx context.Context, deviceCode string) (*DeviceData, error) {
        if err := c.ensureAuth(ctx); err != nil {
                return nil, err
        }

        req := api.NewDeviceDataRequest(deviceCode)
        body, err := c.http.Post(ctx, api.EndpointDeviceData, req, c.auth.FormatTokenHeader())
        if err != nil {
                return nil, &DeviceError{
                        DeviceCode: deviceCode,
                        Operation:  "getDeviceData",
                        Err:        err,
                }
        }

        var resp api.DeviceDataResponse
        if err := json.Unmarshal(body, &resp); err != nil {
                return nil, &DeviceError{
                        DeviceCode: deviceCode,
                        Operation:  "parseDeviceData",
                        Err:        err,
                }
        }

        if !resp.IsSuccess() {
                return nil, &APIError{
                        ErrorCode: resp.ErrorCode,
                        Message:   resp.ErrorMessage,
                }
        }

        parsed := resp.ParseDeviceData()
        data := parsedDataToDeviceData(deviceCode, parsed)
        return &data, nil
}

// Control sends a compound control command to a device.
func (c *client) Control(ctx context.Context, command ControlCommand) error {
        if command.DeviceCode == "" {
                return &ValidationError{
                        Field:   "deviceCode",
                        Value:   command.DeviceCode,
                        Message: "device code is required",
                }
        }

        if command.TargetTemp != nil {
                if err := ValidateTemperature(*command.TargetTemp); err != nil {
                        return err
                }
        }

        if command.Mode != nil {
                if err := ValidateMode(*command.Mode); err != nil {
                        return err
                }
        }

        var values []api.ControlValue

        if command.Power != nil {
                value := api.APIPowerOff
                if *command.Power {
                        value = api.APIPowerOn
                }
                values = append(values, api.ControlValue{ProtocolCode: api.ParamPower, Value: value})
        }

        if command.Mode != nil {
                values = append(values, api.ControlValue{ProtocolCode: api.ParamMode, Value: modeToAPI(*command.Mode)})
        }

        if command.TargetTemp != nil {
                values = append(values, api.ControlValue{ProtocolCode: api.ParamTargetTemp, Value: strconv.FormatFloat(*command.TargetTemp, 'f', 1, 64)})
        }

        if command.FanSpeed != nil {
                values = append(values, api.ControlValue{ProtocolCode: api.ParamFanSpeed, Value: fanSpeedToAPI(*command.FanSpeed)})
        }

        if len(values) == 0 {
                return nil
        }

        if err := c.ensureAuth(ctx); err != nil {
                return err
        }

        for _, v := range values {
                req := api.NewControlRequest(command.DeviceCode, v.ProtocolCode, v.Value)
                body, err := c.http.Post(ctx, api.EndpointControl, req, c.auth.FormatTokenHeader())
                if err != nil {
                        return &DeviceError{
                                DeviceCode: command.DeviceCode,
                                Operation:  "control",
                                Err:        err,
                        }
                }

                var resp api.ControlResponse
                if err := json.Unmarshal(body, &resp); err != nil {
                        return &DeviceError{
                                DeviceCode: command.DeviceCode,
                                Operation:  "parseControlResponse",
                                Err:        err,
                        }
                }

                if !resp.IsSuccess() {
                        return &APIError{
                                ErrorCode: resp.ErrorCode,
                                Message:   resp.ErrorMessage,
                        }
                }
        }

        return nil
}

                        DeviceCode: deviceCode,

                }
        }

        var resp api.ControlResponse
        if err := json.Unmarshal(body, &resp); err != nil {
                return &DeviceError{
                        DeviceCode: deviceCode,
                        Operation:  "parseControlResponse",
                        Err:        err,
                }
        }

        if !resp.IsSuccess() {
                return &APIError{
                        ErrorCode: resp.ErrorCode,
                        Message:   resp.ErrorMessage,
                }
        }

        return nil
}

// ensureAuth ensures we have a valid authentication token.
func (c *client) ensureAuth(ctx context.Context) error {
        if c.auth.IsAuthenticated() {
                return nil
        }

        if err := c.auth.EnsureAuthenticated(ctx); err != nil {
                return &AuthError{
                        Code:      -1,
                        Message:   "not authenticated",
                        Retryable: true,
                        Err:       ErrNotAuthenticated,
                }
        }
        return nil
}

func deviceInfoToDevice(info *api.DeviceInfo) Device {
        return Device{
                Code:        info.DeviceCode,
                Name:        info.DeviceNick,
                Type:        info.DeviceType,
                Online:      info.IsOnline(),
                ProductCode: info.ProductCode,
                CreatedAt:   info.ParsedCreateTime(),
        }
}

func parsedDataToDeviceData(deviceCode string, parsed *api.ParsedDeviceData) DeviceData {
        return DeviceData{
                DeviceCode:  deviceCode,
                Timestamp:   time.Now(),
                Power:       parsed.Power,
                Mode:        parseMode(parsed.Mode),
                TargetTemp:  parsed.TargetTemp,
                CurrentTemp: parsed.CurrentTemp,
                InletTemp:   parsed.InletTemp,
                OutletTemp:  parsed.OutletTemp,
                AmbientTemp: parsed.AmbientTemp,
                CoilTemp:    parsed.CoilTemp,
                FanSpeed:    parseFanSpeed(parsed.FanSpeed),
                ErrorCode:   parsed.ErrorCode,
                RawParams:   parsed.RawParams,
        }
}

func parseMode(value string) OperationMode {
        switch value {
        case api.APIModeOff:
                return ModeOff
        case api.APIModeHeat:
                return ModeHeat
        case api.APIModeCool:
                return ModeCool
        case api.APIModeAuto:
                return ModeAuto
        default:
                return ModeOff
        }
}

func parseFanSpeed(value string) FanSpeed {
        switch value {
        case api.APIFanLow:
                return FanLow
        case api.APIFanMedium:
                return FanMedium
        case api.APIFanHigh:
                return FanHigh
        case api.APIFanAuto:
                return FanAuto
        default:
                return FanAuto
        }
}

func modeToAPI(mode OperationMode) string {

                return api.APIModeHeat
        case ModeCool:
                return api.APIModeCool
        case ModeAuto:
                return api.APIModeAuto

                return api.APIModeOff
        }
}

        case FanHigh:

func getDeviceCode(client goaquatemp.Client) string {
        if deviceCode != "" {
                return deviceCode
        }
        if env := os.Getenv("AQUATEMP_DEFAULT_DEVICE"); env != "" {
                return env
        }
        if client.Config().DefaultDevice != "" {
                return client.Config().DefaultDevice
        }
        return ""
}

func requireDeviceCode(client goaquatemp.Client) (string, error) {
        code := getDeviceCode(client)
        if code == "" {
                return "", fmt.Errorf("device code required: use --device flag or set AQUATEMP_DEFAULT_DEVICE")
        }
        return code, nil
}

func runTempHistory(hours, interval int) error {
        client, err := getClient()
        if err != nil {
                return err
        }

        code, err := requireDeviceCode(client)
        if err != nil {
                return err
        }

        ctx, cancel := context.WithTimeout(context.Background(), time.Duration(hours)*time.Hour+time.Minute)
        defer cancel()

        if err := client.Login(ctx); err != nil {
                return fmt.Errorf("login failed: %w", err)
        }

        endTime := time.Now().Add(time.Duration(hours) * time.Hour)
        ticker := time.NewTicker(time.Duration(interval) * time.Second)
        defer ticker.Stop()

        if outputCSV {
                fmt.Println("timestamp,target,current,inlet,outlet,ambient")
        }

        readAndPrint := func() {
                data, err := client.GetDeviceData(ctx, code)
                if err != nil {
                        fmt.Fprintf(os.Stderr, "error: %v\n", err)
                        return
                }
                now := time.Now().Format("15:04:05")
                if outputJSON {
                        printJSON(map[string]any{
                                "timestamp": now,
                                "target":    data.TargetTemp,
                                "current":   data.CurrentTemp,
                                "inlet":     data.InletTemp,
                                "outlet":    data.OutletTemp,
                                "ambient":   data.AmbientTemp,
                        })
                } else if outputCSV {
                        fmt.Printf("%s,%s,%s,%s,%s,%s\n",
                                now,
                                tempStr(data.TargetTemp),
                                tempStr(data.CurrentTemp),
                                tempStr(data.InletTemp),
                                tempStr(data.OutletTemp),
                                tempStr(data.AmbientTemp),
                        )
                } else {
                        fmt.Printf("[%s] Target: %s°C  Current: %s°C  Inlet: %s°C  Outlet: %s°C  Ambient: %s°C\n",
                                now,
                                tempStr(data.TargetTemp),
                                tempStr(data.CurrentTemp),
                                tempStr(data.InletTemp),
                                tempStr(data.OutletTemp),
                                tempStr(data.AmbientTemp),
                        )
                }
        }

        readAndPrint()

        for {
                select {
                case <-ctx.Done():
                        return nil
                case t := <-ticker.C:
                        if t.After(endTime) {
                                return nil
                        }
                        readAndPrint()
                }
        }
}

func runControlPower(on bool) error {

                return err

        code, err := requireDeviceCode(client)

                return err
        }

// APIVersion represents the version of the Aqua Temp API to use.
type APIVersion string

const (
        // APILegacy is for accounts created before app v1.5.8.
        APILegacy APIVersion = "legacy"
        // APIHiTemp is for HiTemp branded devices.
        APIHiTemp APIVersion = "hitemp"
        // APIModern is for accounts created with app v1.5.9+ (default).
        APIModern APIVersion = "modern"
)

// API base URLs for different versions.
const (
        BaseURLLegacy = "https://cloud.linked-go.com/cloudservice/api"
        BaseURLHiTemp = "https://cloud.linked-go.com/cloudservice/api"
        BaseURLModern = "https://cloud.linked-go.com:449/crmservice/api"
)

// Config holds the configuration for the Aqua Temp client.
type Config struct {
        Username      string        `yaml:"username"`
        Password      string        `yaml:"password"`
        APIVersion    APIVersion    `yaml:"api_version"`
        BaseURL       string        `yaml:"base_url"`
        Timeout       time.Duration `yaml:"timeout"`
        RetryCount    int           `yaml:"retry_count"`
        Debug         bool          `yaml:"debug"`
        DefaultDevice string        `yaml:"default_device"`
        ConfigPath    string        `yaml:"-"`
}

// DefaultConfig returns a Config with default values.
func DefaultConfig() *Config {
        return &Config{
                APIVersion: APIModern,
                Timeout:    30 * time.Second,
                RetryCount: 3,
                Debug:      false,
        }
}

// GetBaseURL returns the base URL for the configured API version.
func (c *Config) GetBaseURL() string {
        if c.BaseURL != "" {
                return c.BaseURL
        }

        switch c.APIVersion {
        case APILegacy:
                return BaseURLLegacy
        case APIHiTemp:
                return BaseURLHiTemp
        case APIModern:
                return BaseURLModern
        default:
                return BaseURLModern
        }
}

// LoadConfigFile loads configuration from a YAML file.
func LoadConfigFile(path string) (*Config, error) {
        // Expand ~ to home directory
        if len(path) > 0 && path[0] == '~' {
                home, err := os.UserHomeDir()
                if err != nil {
                        return nil, err
                }
                path = filepath.Join(home, path[1:])
        }

        data, err := os.ReadFile(path)
        if err != nil {
                return nil, err
        }

        config := DefaultConfig()
        if err := yaml.Unmarshal(data, config); err != nil {
                return nil, err
        }

        return config, nil
}

// LoadConfigFromEnv loads configuration from environment variables.
func LoadConfigFromEnv() *Config {
        config := DefaultConfig()

        if v := os.Getenv("AQUATEMP_USERNAME"); v != "" {
                config.Username = v
        }
        if v := os.Getenv("AQUATEMP_PASSWORD"); v != "" {
                config.Password = v
        }
        if v := os.Getenv("AQUATEMP_API_VERSION"); v != "" {
                config.APIVersion = APIVersion(v)
        }
        if v := os.Getenv("AQUATEMP_BASE_URL"); v != "" {
                config.BaseURL = v
        }
        if v := os.Getenv("AQUATEMP_TIMEOUT"); v != "" {
                if d, err := time.ParseDuration(v); err == nil {
                        config.Timeout = d
                }
        }
        if v := os.Getenv("AQUATEMP_DEBUG"); v != "" {
                config.Debug, _ = strconv.ParseBool(v)
        }
        if v := os.Getenv("AQUATEMP_DEFAULT_DEVICE"); v != "" {
                config.DefaultDevice = v
        }
        if v := os.Getenv("AQUATEMP_CONFIG"); v != "" {
                config.ConfigPath = v
        }

        return config
}

// DefaultConfigPath returns the default config file path.
func DefaultConfigPath() string {
        home, err := os.UserHomeDir()
        if err != nil {
                return ""
        }
        return filepath.Join(home, ".config", "aquatemp", "config.yaml")
}

// Validate checks if the configuration is valid.
func (c *Config) Validate() error {
        if c.Username == "" {
                return &ValidationError{Field: "username", Value: "", Message: "username is required"}
        }
        if c.Password == "" {
                return &ValidationError{Field: "password", Value: "", Message: "password is required"}
        }
        return nil
}

                return &ValidationError{
                        Field:   "temperature",
                        Value:   temp,
                        Message: "temperature must be between 15°C and 40°C",
                        Err:     ErrInvalidTemperature,

        }

// AuthError represents an authentication-related error.
type AuthError struct {
        Code      int
        Message   string
        Retryable bool
        Err       error
}

func (e *AuthError) Error() string {
        if e.Err != nil {
                return fmt.Sprintf("auth error %d: %s: %v", e.Code, e.Message, e.Err)
        }
        return fmt.Sprintf("auth error %d: %s", e.Code, e.Message)
}

func (e *AuthError) Unwrap() error {
        return e.Err
}

// APIError represents an error returned by the Aqua Temp API.
type APIError struct {
        StatusCode int
        ErrorCode  int
        Message    string
}

func (e *APIError) Error() string {
        return fmt.Sprintf("api error %d (http %d): %s", e.ErrorCode, e.StatusCode, e.Message)
}

// DeviceError represents an error related to a specific device operation.
type DeviceError struct {
        DeviceCode string
        Operation  string
        Err        error
}

func (e *DeviceError) Error() string {
        return fmt.Sprintf("device %s: %s failed: %v", e.DeviceCode, e.Operation, e.Err)
}

func (e *DeviceError) Unwrap() error {
        return e.Err
}

// ValidationError represents a validation error for input parameters.
type ValidationError struct {
        Field   string
        Value   any
        Message string
        Err     error
}

func (e *ValidationError) Error() string {
        return fmt.Sprintf("validation error: %s=%v: %s", e.Field, e.Value, e.Message)
}

func (e *ValidationError) Unwrap() error {
        return e.Err
}

// IsAuthError returns true if the error is an authentication error.
func IsAuthError(err error) bool {
        var authErr *AuthError
        return errors.As(err, &authErr)
}

// IsAPIError returns true if the error is an API error.
func IsAPIError(err error) bool {
        var apiErr *APIError
        return errors.As(err, &apiErr)
}

// IsDeviceError returns true if the error is a device error.
func IsDeviceError(err error) bool {
        var deviceErr *DeviceError
        return errors.As(err, &deviceErr)
}

// IsRetryable returns true if the error is retryable.
func IsRetryable(err error) bool {
        var authErr *AuthError
        if errors.As(err, &authErr) {
                return authErr.Retryable
        }

        var apiErr *APIError
        if errors.As(err, &apiErr) {
                // Retry on server errors
                return apiErr.StatusCode >= 500
        }

        return false
}

// APIVersion represents the API version for password handling.
type APIVersion string

const (
        APIModern APIVersion = "modern"
)

// AuthError represents an authentication error.
type AuthError struct {
        Code      int
        Message   string
        Retryable bool
        Err       error
}

func (e *AuthError) Error() string {
        if e.Err != nil {
                return fmt.Sprintf("auth error %d: %s: %v", e.Code, e.Message, e.Err)
        }
        return fmt.Sprintf("auth error %d: %s", e.Code, e.Message)
}

func (e *AuthError) Unwrap() error {
        return e.Err
}

// AuthManager defines the interface for authentication operations.
type AuthManager interface {
        Authenticate(ctx context.Context) (*Token, error)
        RefreshToken(ctx context.Context) (*Token, error)
        GetToken() *Token
        GetTokenValue() string
        IsAuthenticated() bool
        Clear()
        EnsureAuthenticated(ctx context.Context) error
        FormatTokenHeader() string
}

var _ AuthManager = (*Manager)(nil)

// Manager handles authentication with the Aqua Temp API.
type Manager struct {
        store      *TokenStore
        http       transport.HTTPClient
        apiVersion APIVersion
        username   string
        password   string
}

// NewManager creates a new authentication manager.
func NewManager(http transport.HTTPClient, apiVersion string, username, password string) *Manager {
        return &Manager{
                store:      NewTokenStore(),
                http:       http,
                apiVersion: APIVersion(apiVersion),
                username:   username,
                password:   password,
        }
}

// Authenticate performs authentication and stores the token.
func (m *Manager) Authenticate(ctx context.Context) (*Token, error) {
        password := m.preparePassword()

        req := api.NewLoginRequest(m.username, password)

        body, err := m.http.Post(ctx, api.EndpointLogin, req, "")
        if err != nil {
                return nil, &AuthError{
                        Code:      -1,
                        Message:   "failed to send login request",
                        Retryable: true,
                        Err:       err,
                }
        }

        var resp api.LoginResponse
        if err := json.Unmarshal(body, &resp); err != nil {
                return nil, &AuthError{
                        Code:      -1,
                        Message:   "failed to parse login response",
                        Retryable: false,
                        Err:       err,
                }
        }

        if !resp.IsSuccess() {
                retryable := resp.ErrorCode >= 500
                return nil, &AuthError{
                        Code:      resp.ErrorCode,
                        Message:   resp.ErrorMessage,
                        Retryable: retryable,
                }
        }

        if resp.ObjectResult == nil || resp.ObjectResult.Token == "" {
                return nil, &AuthError{
                        Code:    -1,
                        Message: "no token in response",
                }
        }

        expiresIn := resp.ObjectResult.ExpiresIn
        if expiresIn <= 0 {
                expiresIn = 3600 // Default to 1 hour
        }

        token := &Token{
                Value:     resp.ObjectResult.Token,
                UserID:    resp.ObjectResult.UserID,
                ExpiresAt: time.Now().Add(time.Duration(expiresIn) * time.Second),
        }

        m.store.Set(token)
        return token, nil
}

// RefreshToken refreshes the authentication token.
func (m *Manager) RefreshToken(ctx context.Context) (*Token, error) {
        // The Aqua Temp API doesn't have a refresh endpoint,
        // so we just re-authenticate.
        return m.Authenticate(ctx)
}

// GetToken returns the current token.
func (m *Manager) GetToken() *Token {
        return m.store.Get()
}

// GetTokenValue returns the current token value string.
func (m *Manager) GetTokenValue() string {
        token := m.store.Get()
        if token == nil {
                return ""
        }
        return token.Value
}

// IsAuthenticated returns true if we have a valid token.
func (m *Manager) IsAuthenticated() bool {
        return m.store.IsValid()
}

// Clear removes the stored token.
func (m *Manager) Clear() {
        m.store.Clear()
}

// EnsureAuthenticated ensures we have a valid token, refreshing if needed.
func (m *Manager) EnsureAuthenticated(ctx context.Context) error {
        if m.IsAuthenticated() {
                return nil
        }

        _, err := m.RefreshToken(ctx)
        return err
}

// preparePassword prepares the password for the API.
// Modern API (v1.5.9+) requires MD5 hashed password.
func (m *Manager) preparePassword() string {
        if m.apiVersion == APIModern {
                return md5Hash(m.password)
        }
        return m.password
}

// md5Hash returns the MD5 hash of a string as a lowercase hex string.
func md5Hash(s string) string {
        hash := md5.Sum([]byte(s))
        return strings.ToLower(hex.EncodeToString(hash[:]))
}

// FormatTokenHeader returns the token formatted for the x-token header.
func (m *Manager) FormatTokenHeader() string {
        token := m.GetTokenValue()
        if token == "" {
                return ""
        }
        return fmt.Sprintf("bearer %s", token)
}

// Token represents an authentication token.
type Token struct {
        Value     string
        UserID    string
        ExpiresAt time.Time
}

// IsValid returns true if the token is set and not expired.
func (t *Token) IsValid() bool {
        if t == nil || t.Value == "" {
                return false
        }
        // Add a buffer of 60 seconds before actual expiry
        return time.Now().Add(60 * time.Second).Before(t.ExpiresAt)
}

// IsExpired returns true if the token has expired.
func (t *Token) IsExpired() bool {
        if t == nil {
                return true
        }
        return time.Now().After(t.ExpiresAt)
}

// TokenStore provides thread-safe storage for authentication tokens.
type TokenStore struct {
        mu    sync.RWMutex
        token *Token
}

// NewTokenStore creates a new TokenStore.
func NewTokenStore() *TokenStore {
        return &TokenStore{}
}

// Set stores a token.
func (s *TokenStore) Set(token *Token) {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.token = token
}

// Get returns the current token.
func (s *TokenStore) Get() *Token {
        s.mu.RLock()
        defer s.mu.RUnlock()
        return s.token
}

// Clear removes the stored token.
func (s *TokenStore) Clear() {
        s.mu.Lock()
        defer s.mu.Unlock()
        s.token = nil
}

// IsValid returns true if the stored token is valid.
func (s *TokenStore) IsValid() bool {
        s.mu.RLock()
        defer s.mu.RUnlock()
        return s.token.IsValid()
}

// HTTPError represents an HTTP-level error.
type HTTPError struct {
        StatusCode int
        Message    string
}

func (e *HTTPError) Error() string {
        return fmt.Sprintf("http error %d: %s", e.StatusCode, e.Message)
}

// HTTPClient defines the interface for making HTTP requests to the API.
type HTTPClient interface {
        Post(ctx context.Context, endpoint string, body any, token string) ([]byte, error)
        Get(ctx context.Context, endpoint string, token string) ([]byte, error)
        SetBaseURL(baseURL string)
}

var _ HTTPClient = (*Client)(nil)

// Client wraps the HTTP client for API requests.
type Client struct {
        baseURL    string
        httpClient *http.Client
        debug      bool
        retryCount int
}

// NewClient creates a new HTTP transport client.
func NewClient(baseURL string, timeout time.Duration, retryCount int, debug bool) *Client {
        return &Client{
                baseURL: baseURL,
                httpClient: &http.Client{
                        Timeout: timeout,
                },
                debug:      debug,
                retryCount: retryCount,
        }
}

// Post sends a POST request to the API.
func (c *Client) Post(ctx context.Context, endpoint string, body any, token string) ([]byte, error) {
        return c.doRequest(ctx, http.MethodPost, endpoint, body, token)
}

// Get sends a GET request to the API.
func (c *Client) Get(ctx context.Context, endpoint string, token string) ([]byte, error) {
        return c.doRequest(ctx, http.MethodGet, endpoint, nil, token)
}

func (c *Client) doRequest(ctx context.Context, method, endpoint string, body any, token string) ([]byte, error) {
        url := c.baseURL + endpoint

        var bodyReader io.Reader
        if body != nil {
                jsonBody, err := json.Marshal(body)
                if err != nil {
                        return nil, fmt.Errorf("failed to marshal request body: %w", err)
                }
                bodyReader = bytes.NewReader(jsonBody)

                if c.debug {
                        log.Printf("[DEBUG] Request: %s %s\n%s", method, url, string(jsonBody))
                }
        }

        var lastErr error
        for attempt := 0; attempt <= c.retryCount; attempt++ {
                if attempt > 0 {
                        // Exponential backoff
                        backoff := time.Duration(attempt*attempt) * 100 * time.Millisecond
                        select {
                        case <-ctx.Done():
                                return nil, ctx.Err()
                        case <-time.After(backoff):
                        }

                        // Re-create body reader for retry
                        if body != nil {
                                jsonBody, _ := json.Marshal(body)
                                bodyReader = bytes.NewReader(jsonBody)
                        }
                }

                req, err := http.NewRequestWithContext(ctx, method, url, bodyReader)
                if err != nil {
                        return nil, fmt.Errorf("failed to create request: %w", err)
                }

                req.Header.Set("Content-Type", "application/json")
                req.Header.Set("Accept", "application/json")
                req.Header.Set("User-Agent", "AquaTemp-Go/1.0")

                if token != "" {
                        req.Header.Set("x-token", token)
                }

                resp, err := c.httpClient.Do(req)
                if err != nil {
                        lastErr = err
                        if c.debug {
                                log.Printf("[DEBUG] Request failed (attempt %d): %v", attempt+1, err)
                        }
                        continue
                }

                respBody, err := io.ReadAll(resp.Body)
                resp.Body.Close()

                if err != nil {
                        lastErr = err
                        continue
                }

                if c.debug {
                        log.Printf("[DEBUG] Response (%d): %s", resp.StatusCode, string(respBody))
                }

                if resp.StatusCode >= 500 {
                        lastErr = &HTTPError{
                                StatusCode: resp.StatusCode,
                                Message:    fmt.Sprintf("server error: %s", http.StatusText(resp.StatusCode)),
                        }
                        continue
                }

                if resp.StatusCode >= 400 {
                        return nil, &HTTPError{
                                StatusCode: resp.StatusCode,
                                Message:    fmt.Sprintf("client error: %s", http.StatusText(resp.StatusCode)),
                        }
                }

                return respBody, nil
        }

        if lastErr != nil {
                return nil, fmt.Errorf("request failed after %d attempts: %w", c.retryCount+1, lastErr)
        }

        return nil, fmt.Errorf("request failed")
}

// SetBaseURL updates the base URL.
func (c *Client) SetBaseURL(baseURL string) {
        c.baseURL = baseURL
}

                if cfg.Password != "" {
                        c.Password = cfg.Password
                }
                if cfg.APIVersion != "" {
                        c.APIVersion = cfg.APIVersion
                }
                if cfg.BaseURL != "" {
                        c.BaseURL = cfg.BaseURL
                }
                if cfg.Timeout > 0 {
                        c.Timeout = cfg.Timeout
                }
                if cfg.RetryCount > 0 {
                        c.RetryCount = cfg.RetryCount
                }
                c.Debug = cfg.Debug
                if cfg.DefaultDevice != "" {
                        c.DefaultDevice = cfg.DefaultDevice
                }
        }
}

// WithConfigFile loads configuration from a file and applies it.
func WithConfigFile(path string) Option {
        return func(c *Config) {
                cfg, err := LoadConfigFile(path)
                if err != nil {
                        return // Silently ignore file load errors
                }
                WithConfig(cfg)(c)
        }
}

// WithEnvConfig loads configuration from environment variables.
func WithEnvConfig() Option {
        return func(c *Config) {
                cfg := LoadConfigFromEnv()
                WithConfig(cfg)(c)
        }
}
